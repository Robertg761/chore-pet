from pathlib import Path
import subprocess
sources=['src/screens/CharacterCreator.css','scripts/browser/creator-thumbnail.test.mjs','docs/ART.md','docs/audits/2026-10-12-character-polish/creator-thumbnail-design.md']
for p in sources:
 tracked=subprocess.run(['git','ls-files','--error-unmatch',p],capture_output=True).returncode==0
 r=subprocess.run(['git','diff','--check','--',p] if tracked else ['git','diff','--no-index','--check','--','/dev/null',p],capture_output=True,text=True)
 assert r.returncode==(0 if tracked else 1) and not(r.stdout+r.stderr).strip(),(p,r)
 print(p,'whitespace pass')
