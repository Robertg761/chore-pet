import { readFileSync } from 'node:fs'
const candidates = {
  '/src/screens/WardrobePreview.tsx': '/tmp/wardrobe-mirror-WardrobePreview.tsx',
  '/src/screens/Wardrobe.css': '/tmp/wardrobe-mirror-Wardrobe.css',
}
export default {
  plugins: [{ name: 'isolated-wardrobe-mirror-candidate', enforce: 'pre', transform(code, id) {
    const path = id.split('?')[0], candidate = candidates[path.slice(process.cwd().length)]
    if (candidate) return { code: readFileSync(candidate, 'utf8'), map: null }
  } }],
  test: { environment: 'node', include: ['src/**/*.test.{ts,tsx}', 'supabase/**/*.test.ts'] },
}
