def point(r, angle):
    return (r * math.cos(angle), r * math.sin(angle))


def fmt(p):
    return ' '.join(f'{v:.6f}'.rstrip('0').rstrip('.') for v in p)


def petal_path(valley, tip=17, width=11):
    parts = []
    radial = (tip - .25 * valley * math.cos(math.pi / 5)) / .75
    for i in range(5):
        a = -math.pi / 2 + i * 2 * math.pi / 5
        start, end = point(valley, a - math.pi / 5), point(valley, a + math.pi / 5)
        center = point(radial, a)
        tangent = (-math.sin(a), math.cos(a))
        c1 = (center[0] - width * tangent[0], center[1] - width * tangent[1])
        c2 = (center[0] + width * tangent[0], center[1] + width * tangent[1])
        if not i: parts.append('M' + fmt(start))
        parts.append('C' + fmt(c1) + ' ' + fmt(c2) + ' ' + fmt(end))
    return ' '.join(parts) + ' Z'

def flower(valley=10, stroke_scale=1, tip=17, width=11, center=5.6):
    return f'''<g stroke="{ink}" stroke-width="{4 * stroke_scale}" stroke-linejoin="round" stroke-linecap="round">
      <path d="{leaf}" fill="{palette['leaf']}" stroke="none"/>
      <path d="{shade}" fill="{palette['leafDark']}" stroke="none"/>
      <path d="{leaf}" fill="none"/>
      <path d="{petal_path(valley, tip, width)}" fill="{palette['cream']}"/>
      <circle cx="0" cy="0" r="{center}" fill="{palette['petDefault']}" stroke-width="{2.5 * stroke_scale}"/>
      <circle cx="-1.3" cy="-1.5" r="1" fill="{palette['white']}" stroke="none"/>
    </g>'''

def head(name, color, stroke_scale=1):
    appendage = ''
    if name == 'bun':
        e = f'<g transform="rotate(-6 82 98)"><path d="{ear}" fill="{color}"/></g>'
        appendage = e + f'<g transform="translate(200 0) scale(-1 1)">{e}</g>'
    elif name == 'mochi':
        appendage = f'<path d="M89 78 C88 67 94 60 100 60 C106 60 112 67 111 78 Z" fill="{color}"/><path d="M100 60 C100 55 104 53 106 55" fill="none" stroke-width="{3.5 * stroke_scale}"/>'
    else:
        appendage = f'''<path d="M100 74 C100 62 98 54 101 46" fill="none" stroke-width="{4 + 7 * stroke_scale}"/><path d="M100 74 C100 62 98 54 101 46" fill="none" stroke="{palette['leaf']}" stroke-width="4"/>
          <path d="M101 46 C94 32 78 26 64 32 C70 46 86 52 101 46 Z" fill="{palette['leaf']}"/>
          <path d="M101 46 C108 30 126 22 142 28 C136 44 118 52 101 46 Z" fill="{palette['leaf']}"/>'''
    x, y, scale = anchors[name]
    px, py, angle = placements[name]
    return f'<g stroke="{ink}" stroke-width="{4 * stroke_scale}" stroke-linejoin="round" stroke-linecap="round">{appendage}<path d="{bodies[name]}" fill="{color}"/></g><g transform="translate({x} {y}) scale({scale}) translate({px} {py}) rotate({angle})">{flower(10, stroke_scale)}</g>'
