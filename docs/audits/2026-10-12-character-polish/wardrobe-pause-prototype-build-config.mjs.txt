import base from '/home/robert/.codex/worktrees/4511/chore-pet/vite.config.ts'
import { readFileSync } from 'node:fs'
const candidates = {
  '/src/screens/WardrobePreview.tsx': '/tmp/wardrobe-pause-WardrobePreview.tsx',
  '/src/screens/Wardrobe.css': '/tmp/wardrobe-pause-Wardrobe.css',
}
export default options => {
  const config = base(options)
  const candidate = { name: 'isolated-wardrobe-pause-build', enforce: 'pre', transform(code, id) {
    const path = id.split('?')[0], source = candidates[path.slice(process.cwd().length)]
    if (source) return { code: readFileSync(source, 'utf8'), map: null }
  } }
  return { ...config, plugins: [candidate, ...config.plugins] }
}
