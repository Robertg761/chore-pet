// @vitest-environment jsdom
import { act, StrictMode } from 'react'
import { afterEach, beforeEach, expect, it, vi } from 'vitest'
import type { Mood, Pet, Species } from '../domain/types'
import { setPetMotionReduced } from '../pet/motionPreference'
import { click, render } from '../test/react'
import { CharacterCreator, type PetLookPatch } from './CharacterCreator'

const pet: Pet = { id: 'pet', homeId: 'home', name: 'Mochi', species: 'mochi', bodyColour: '#FFCFDA', eyes: 'lashes', cheeks: 'hearts', equipped: { head: 'beanie-red', face: 'glasses', neck: 'scarf', outfit: 'overalls', back: 'backpack' } }
let ui: ReturnType<typeof render>
let now: number, serial: number
let frames: Map<number, FrameRequestCallback>
let media: MediaQueryList, wide: MediaQueryList
let saved: ReturnType<typeof vi.fn<(patch: PetLookPatch) => void>>
const closed = vi.fn()
const content = (p = pet) => <StrictMode><CharacterCreator pet={p} onSave={saved} onClose={closed} /></StrictMode>

beforeEach(() => {
  setPetMotionReduced(false)
  now = 0; serial = 0; frames = new Map(); saved = vi.fn(); closed.mockClear()
  vi.spyOn(performance, 'now').mockImplementation(() => now)
  media = Object.assign(new EventTarget(), { matches: false }) as MediaQueryList
  wide = Object.assign(new EventTarget(), { matches: false }) as MediaQueryList
  vi.stubGlobal('matchMedia', (query: string) => query.includes('prefers-reduced-motion') ? media : wide)
  vi.stubGlobal('requestAnimationFrame', (callback: FrameRequestCallback) => { frames.set(++serial, callback); return serial })
  vi.stubGlobal('cancelAnimationFrame', (id: number) => frames.delete(id))
  ui = render(content())
})
afterEach(() => { ui.unmount(); setPetMotionReduced(false); vi.restoreAllMocks(); vi.unstubAllGlobals() })

function button(name: string) {
  const node = [...ui.container.querySelectorAll('button')].find(node => node.textContent === name)
  expect(node, `button ${name}`).toBeTruthy(); return node!
}
const art = () => ui.container.querySelector('.cc-preview .ch-animated')!
const image = () => ui.container.querySelector('.cc-preview svg') as SVGSVGElement
function radio(name: string) { return ui.container.querySelector(`[role="radio"][aria-label="${name}"]`)! }
function mood(value: Mood) {
  const node = ui.container.querySelector('.cc-mood select') as HTMLSelectElement
  act(() => { node.value = value; node.dispatchEvent(new Event('change', { bubbles: true })) })
}
function frame(at: number) { now = at; const pending = [...frames.values()]; frames.clear(); act(() => pending.forEach(fn => fn(at))) }
function expectFacing(mirrored: boolean) {
  const group = ui.container.querySelector('[data-preview-facing]')
  expect(group).not.toBeNull()
  expect(group!.getAttribute('data-preview-facing')).toBe(mirrored ? 'mirrored' : 'normal')
  expect(group!.getAttribute('transform')).toBe(mirrored ? 'translate(200 0) scale(-1 1)' : null)
  expect(button('Mirror preview').getAttribute('aria-pressed')).toBe(String(mirrored))
}

it.each((['mochi', 'bun', 'sprout'] as Species[]).flatMap(species => (['happy', 'content', 'meh', 'scruffy', 'sick'] as Mood[]).map(value => [species, value] as const)))('inspects the %s %s face without changing art, saving or starting a clock', (species, value) => {
  ui.unmount(); ui = render(content({ ...pet, species }))
  click(button('Face')); mood(value)
  const node = image(), figure = art(), markup = figure.innerHTML
  const detail = button('Face detail')
  expect(detail.getAttribute('type')).toBe('button'); expect(detail.getAttribute('aria-pressed')).toBe('false')
  expect(node.getAttribute('viewBox')).toBe('0 0 200 200')
  act(() => detail.focus()); click(detail)
  expect(document.activeElement).toBe(detail)
  expect(detail.getAttribute('aria-pressed')).toBe('true')
  expect(node.getAttribute('viewBox')).toBe('40 75 120 110')
  expect(node.getAttribute('aria-label')).toContain('face detail')
  expect(image()).toBe(node); expect(art()).toBe(figure); expect(art().innerHTML).toBe(markup)
  click(button('Mirror preview')); expectFacing(true)
  expect(art().innerHTML).toBe(markup); expect(image().getAttribute('viewBox')).toBe('40 75 120 110')
  click(detail)
  expect(node.getAttribute('viewBox')).toBe('0 0 200 200')
  expect(node.getAttribute('aria-label')).not.toContain('face detail')
  expect(detail.getAttribute('aria-pressed')).toBe('false')
  expect(art().innerHTML).toBe(markup); expect(frames.size).toBe(0)
  expect(saved).not.toHaveBeenCalled(); expect(closed).not.toHaveBeenCalled()
})

it('changes detail and reveals a running cheer without changing its sampled frame or original deadline', () => {
  click(radio('Bun')); frame(756)
  const node = art(), markup = node.innerHTML
  expect(frames.size).toBe(1)
  click(button('Face')); click(button('Face detail')); click(button('Mirror preview'))
  const scroll = vi.fn(), focus = vi.fn(); Object.assign(image(), { scrollIntoView: scroll, focus })
  click(button('Show preview'))
  expect(scroll).toHaveBeenCalledExactlyOnceWith({ behavior: 'instant', block: 'center', inline: 'nearest' })
  expect(focus).toHaveBeenCalledExactlyOnceWith({ preventScroll: true })
  expect(art()).toBe(node); expect(art().innerHTML).toBe(markup); expect(frames.size).toBe(1)
  expect(image().getAttribute('viewBox')).toBe('40 75 120 110')
  frame(1799); expect(frames.size).toBe(1)
  frame(1800); expect(frames.size).toBe(0)
  expect(art().querySelector('[data-expression="happy"]')).not.toBeNull(); expectFacing(true)
  expect(button('Face detail').getAttribute('aria-pressed')).toBe('true')
  expect(saved).not.toHaveBeenCalled(); expect(closed).not.toHaveBeenCalled()
})

it('keeps detail local through mood, tab, resize and draft edits, then resets on reopening', () => {
  click(button('Face')); click(button('Face detail')); click(button('Mirror preview')); mood('meh')
  click(button('Who')); click(radio('Bun'))
  act(() => { Object.defineProperty(wide, 'matches', { value: true, configurable: true }); wide.dispatchEvent(new Event('change')) })
  click(radio('Sparkly')); click(radio('Hearts'))
  expectFacing(true); expect(button('Face detail').getAttribute('aria-pressed')).toBe('true')
  expect(image().getAttribute('viewBox')).toBe('40 75 120 110')
  expect(image().getAttribute('aria-label')).toContain('Tired'); expect(frames.size).toBe(0)
  click(button('Save'))
  expect(saved).toHaveBeenCalledExactlyOnceWith({ species: 'bun', bodyColour: '#FFFFFF', eyes: 'sparkly' })
  ui.unmount(); ui = render(content())
  expectFacing(false); expect(button('Face detail').getAttribute('aria-pressed')).toBe('false')
  expect(image().getAttribute('viewBox')).toBe('0 0 200 200'); expect(frames.size).toBe(0)
})

it.each(['system', 'device'] as const)('keeps detail available under live %s reduction without replaying the cheer', policy => {
  click(radio('Bun')); frame(756); click(button('Face')); click(button('Face detail'))
  const reduce = (value: boolean) => act(() => {
    if (policy === 'device') setPetMotionReduced(value)
    else { Object.defineProperty(media, 'matches', { value, configurable: true }); media.dispatchEvent(new Event('change')) }
  })
  reduce(true); expect(frames.size).toBe(0)
  const markup = art().innerHTML
  click(button('Face detail')); click(button('Face detail'))
  expect(art().innerHTML).toBe(markup); expect(frames.size).toBe(0)
  reduce(false); frame(5000)
  expect(art().innerHTML).toBe(markup); expect(frames.size).toBe(0)
  expect(image().getAttribute('viewBox')).toBe('40 75 120 110')
  expect(saved).not.toHaveBeenCalled(); expect(closed).not.toHaveBeenCalled()
})

it.each(['system', 'device'] as const)('starts with static detail available under initial %s reduction', policy => {
  ui.unmount()
  if (policy === 'device') setPetMotionReduced(true)
  else Object.defineProperty(media, 'matches', { value: true, configurable: true })
  ui = render(content()); click(button('Face'))
  const write = vi.spyOn(Storage.prototype, 'setItem')
  click(button('Face detail')); click(button('Mirror preview')); mood('sick')
  expect(image().getAttribute('viewBox')).toBe('40 75 120 110'); expectFacing(true)
  expect(frames.size).toBe(0); expect(saved).not.toHaveBeenCalled(); expect(write).not.toHaveBeenCalled()
})
