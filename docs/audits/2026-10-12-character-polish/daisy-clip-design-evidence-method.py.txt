def png_info(p):
 data=p.read_bytes();assert data[:8]==b'\x89PNG\r\n\x1a\n',p
 offset=8;compressed=[];header=None;ended=False
 while offset<len(data):
  n=struct.unpack('>I',data[offset:offset+4])[0];kind=data[offset+4:offset+8];payload=data[offset+8:offset+8+n]
  assert len(payload)==n and len(data)>=offset+n+12
  assert zlib.crc32(kind+payload)&0xffffffff==struct.unpack('>I',data[offset+n+8:offset+n+12])[0]
  if kind==b'IHDR':header=struct.unpack('>IIBBBBB',payload)
  if kind==b'IDAT':compressed.append(payload)
  offset+=n+12
  if kind==b'IEND':ended=True;break
 assert ended and offset==len(data)
 w,h,depth,colour,compression,filtering,interlace=header
 assert depth==8 and colour in (2,6) and compression==filtering==interlace==0
 raw=zlib.decompress(b''.join(compressed));assert len(raw)==h*(1+w*(4 if colour==6 else 3))
 return {'width':w,'height':h,'bytes':len(data),'sha256':sha(p),'chunkCrcsAndScanlineLengthValid':True}
