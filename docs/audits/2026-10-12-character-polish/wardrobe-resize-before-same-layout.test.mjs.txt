import assert from 'node:assert/strict'
import { test } from 'node:test'
import { mkdir, writeFile } from 'node:fs/promises'
import { join } from 'node:path'
import { browserApp, snapshot } from './helpers.mjs'

const out = process.env.CHORE_RESIZE_CAPTURE_DIR
async function fixture(t, width, height, text) {
  const page = await browserApp(t, { viewport: { width, height }, reducedMotion: 'no-preference' })
  await page.evaluate(() => new Promise((resolve, reject) => {
    const request = indexedDB.open('chore-pet', 1)
    request.onerror = () => reject(request.error)
    request.onsuccess = () => {
      const db = request.result, tx = db.transaction('kv', 'readwrite'), store = tx.objectStore('kv'), read = store.get('snapshot')
      read.onsuccess = () => {
        const saved = read.result, pet = Object.values(saved.tables.pets)[0]
        pet.equipped = { head: 'beanie-red', face: 'glasses', neck: 'scarf', outfit: 'overalls', back: 'backpack' }
        pet.outfits = [
          { id: 'weekend', name: 'Weekend', equipped: { head: 'bow', outfit: 'hoodie' } },
          { id: 'cosy', name: 'Cosy', equipped: { ...pet.equipped } },
          { id: 'party', name: 'Party clothes', equipped: { head: 'crown', outfit: 'dress' } },
        ]
        const progress = Object.values(saved.tables.progress)[0]
        progress.unlockedItems = [...new Set([...progress.unlockedItems, ...[
          'beanie-red', 'bow', 'glasses', 'scarf', 'bow-tie', 'backpack', 'hoodie', 'overalls',
          'dress', 'knit-sweater', 'leaf-crown', 'chef-hat', 'heart-glasses', 'bandana', 'apron', 'crown',
        ].map(id => `item:${id}`)])]
        saved.outbox = {}; store.put(saved, 'snapshot')
      }
      tx.oncomplete = () => { db.close(); resolve() }
      tx.onerror = () => { db.close(); reject(tx.error) }
    }
  }))
  await page.reload()
  await page.getByRole('button', { name: 'Wardrobe', exact: true }).click()
  await page.getByRole('heading', { name: 'Dress up Mochi' }).waitFor()
  await page.waitForFunction(() => !document.documentElement.dataset.vt)
  await page.evaluate(text => { document.documentElement.style.fontSize = `${text}%` }, text)
  await page.clock.install(); await page.clock.pauseAt(Date.now() + 1000)
  await page.evaluate(() => {
    const request = requestAnimationFrame, cancel = cancelAnimationFrame
    window.__renameFrames = new Set()
    window.requestAnimationFrame = callback => {
      const id = request(now => { window.__renameFrames.delete(id); callback(now) })
      window.__renameFrames.add(id); return id
    }
    window.cancelAnimationFrame = id => { window.__renameFrames.delete(id); cancel(id) }
  })
  const savedTab = page.getByRole('tab', { name: 'Saved', exact: true })
  if (await savedTab.count()) await savedTab.click()
  await page.clock.runFor(240)
  return page
}

async function capture(page, name, data) {
  if (!out) return
  await mkdir(out, { recursive: true })
  await page.screenshot({ path: join(out, `${name}.png`), animations: 'disabled' })
  await writeFile(join(out, `${name}.json`), JSON.stringify(data, null, 2) + '\n')
}

async function layout(page) {
  return page.evaluate(() => {
    const rect = node => { const b = node.getBoundingClientRect(); return { x: b.x, y: b.y, width: b.width, height: b.height, right: b.right, bottom: b.bottom } }
    const focused = document.activeElement, focusBox = rect(focused)
    const hit = document.elementFromPoint(focusBox.x + focusBox.width / 2, focusBox.y + focusBox.height / 2)
    const outfitName = focused.closest('.wd-outfit')?.querySelector('.wd-outfit-name')
    const nameBox = outfitName ? rect(outfitName) : null
    const nameHit = nameBox ? document.elementFromPoint(nameBox.x + nameBox.width / 2, nameBox.y + nameBox.height / 2) : null
    return {
      width: innerWidth, overflow: Math.max(document.documentElement.scrollWidth, document.body.scrollWidth) - innerWidth,
      frames: window.__renameFrames.size,
      controls: [...document.querySelectorAll('.wd-outfit button, .wd-name-form .btn')].map(node => ({ name: node.getAttribute('aria-label') || node.textContent, box: rect(node), parent: rect(node.closest('.wd-outfit') || node.closest('.wd-name-form')) })),
      input: document.querySelector('.wd-name-form input') ? rect(document.querySelector('.wd-name-form input')) : null,
      readableFieldWidth: document.querySelector('.wd-name-form') ? Math.min(document.querySelector('.wd-name-form').getBoundingClientRect().width, 10 * parseFloat(getComputedStyle(document.documentElement).fontSize)) : null,
      focus: document.activeElement?.getAttribute('aria-label') || document.activeElement?.tagName,
      focusedPaintVisible: focused === hit || focused.contains(hit),
      focusedOutfitNamePaintVisible: outfitName ? outfitName === nameHit || outfitName.contains(nameHit) : null,
    }
  })
}
function assertLayout(data) {
  assert.equal(data.focusedPaintVisible, true, 'the focused input or returned Rename button must be visible through actual scroll clipping')
  if (data.focusedOutfitNamePaintVisible !== null) assert.equal(data.focusedOutfitNamePaintVisible, true, 'the saved name must remain readable beside the returned Rename focus')
  assert.ok(data.overflow <= 1, 'enlarged text must not cause horizontal page overflow')
  for (const control of data.controls) {
    assert.ok(control.box.width >= 44 && control.box.height >= 44, `${control.name} retains a 44px target`)
    assert.ok(control.box.x >= control.parent.x - 1 && control.box.right <= control.parent.right + 1, `${control.name} stays within its card or form`)
  }
  if (data.input) assert.ok(data.input.x >= -1 && data.input.right <= data.width + 1, 'name field stays within the viewport')
  if (data.input) assert.ok(data.input.width >= data.readableFieldWidth - 1, 'the name field keeps readable space as text grows')
}

async function resize(page, next) {
  const query = '(min-width: 900px), (orientation: landscape) and (max-height: 500px) and (min-width: 640px)'
  await page.evaluate(query => { const media = matchMedia(query); window.__wideChanged = new Promise(resolve => media.addEventListener('change', () => resolve(), { once: true })) }, query)
  await page.setViewportSize(next)
  let timeout
  try { await Promise.race([page.evaluate(() => window.__wideChanged), new Promise((_, reject) => { timeout = setTimeout(() => reject(new Error('native wide-layout change event missing')), 4000) })]) }
  finally { clearTimeout(timeout) }
}

for (const text of [100, 200]) for (const wide of [false, true]) for (const target of ['input', 'Save name', 'Not now']) test(`keeps ${target} editing through native layout change wide=${wide}/${text}%`, async t => {
  const page = await fixture(t, wide ? 844 : 390, wide ? 390 : 664, text), before = (await snapshot(page)).tables
  await page.getByRole('button', { name: 'Rest', exact: true }).click(); await page.clock.runFor(800)
  await page.getByRole('button', { name: 'Pause movement preview of Mochi', exact: true }).click()
  await page.getByRole('button', { name: 'Mirror preview of Mochi', exact: true }).click()
  const art = page.locator('.wd-preview .ch-animated'), held = await art.evaluate(node => { window.__resizeArt = node; return node.outerHTML })
  await page.getByRole('button', { name: 'Rename Weekend', exact: true }).click()
  const field = page.getByRole('textbox', { name: 'Outfit name', exact: true })
  await field.fill('Sunday outfit'); await field.evaluate(node => { node.focus(); node.setSelectionRange(2, 6, 'backward') })
  if (target !== 'input') await page.getByRole('button', { name: target, exact: true }).focus()
  await resize(page, { width: wide ? 390 : 844, height: wide ? 664 : 390 })
  const focus = target === 'input' ? field : page.getByRole('button', { name: target, exact: true })
  const view = await layout(page)
  await capture(page, `wardrobe-resize-${target.replaceAll(' ', '-')}-${wide ? 'wide' : 'narrow'}-${text}`, { layout: view, target, fromWide: wide, method: 'Native matchMedia change after actual viewport resize, dirty name, focused native control, held mirrored Rest. Root font percentages, not browser zoom.' })
  assert.equal(await field.count(), 1, 'open editor remains available when the layout changes')
  assert.equal(await field.inputValue(), 'Sunday outfit')
  assert.equal(await focus.evaluate(node => node === document.activeElement), true, 'native typing/control focus survives layout change')
  if (target === 'input') assert.deepEqual(await field.evaluate(node => [node.selectionStart, node.selectionEnd, node.selectionDirection]), [2, 6, 'backward'])
  assertLayout(view); assert.equal(view.frames, 0)
  assert.equal(await art.evaluate(node => node === window.__resizeArt), true); assert.equal(await art.evaluate(node => node.outerHTML), held)
  assert.deepEqual((await snapshot(page)).tables, before)
  if (target === 'Not now') {
    await page.keyboard.press('Space'); assert.equal(await field.count(), 0); assert.deepEqual((await snapshot(page)).tables, before)
  } else {
    await page.keyboard.press('Enter')
    const expected = structuredClone(before); Object.values(expected.pets)[0].outfits[0].name = 'Sunday outfit'
    assert.deepEqual((await snapshot(page)).tables, expected)
  }
})

for (const wide of [false, true]) test(`keeps editor draft without stealing preview focus on native resize wide=${wide}`, async t => {
  const page = await fixture(t, wide ? 844 : 390, wide ? 390 : 664, 100), before = (await snapshot(page)).tables
  await page.getByRole('button', { name: 'Rename Weekend', exact: true }).click()
  const field = page.getByRole('textbox', { name: 'Outfit name', exact: true }); await field.fill('Sunday outfit')
  const mirror = page.getByRole('button', { name: 'Mirror preview of Mochi', exact: true }); await mirror.focus()
  await resize(page, { width: wide ? 390 : 844, height: wide ? 664 : 390 })
  assert.equal(await field.count(), 1); assert.equal(await field.inputValue(), 'Sunday outfit')
  assert.equal(await mirror.evaluate(node => node === document.activeElement), true)
  assert.equal(await page.evaluate(() => window.__renameFrames.size), 0); assert.deepEqual((await snapshot(page)).tables, before)
})
