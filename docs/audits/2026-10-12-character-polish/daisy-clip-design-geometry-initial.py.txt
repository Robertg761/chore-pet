def bezier(p,t):
 return (1-t)**3*p[0]+3*(1-t)**2*t*p[1]+3*(1-t)*t*t*p[2]+t**3*p[3]


def extrema(p):
 a=-p[0]+3*p[1]-3*p[2]+p[3];b=2*(p[0]-2*p[1]+p[2]);c=p[1]-p[0]
 roots=[]
 if abs(a)<1e-14:
  if abs(b)>1e-14:roots=[-c/b]
 else:
  d=b*b-4*a*c
  if d>=0:roots=[(-b-math.sqrt(d))/(2*a),(-b+math.sqrt(d))/(2*a)]
 return [bezier(p,t) for t in [0,1,*[r for r in roots if 0<r<1]]]


def bounds(transformed,stroke):
 xs=[];ys=[]
 for curve in transformed:
  xs+=extrema([p[0] for p in curve]);ys+=extrema([p[1] for p in curve])
 return {'left':min(xs)-stroke/2,'right':max(xs)+stroke/2,'top':min(ys)-stroke/2,'bottom':max(ys)+stroke/2}
