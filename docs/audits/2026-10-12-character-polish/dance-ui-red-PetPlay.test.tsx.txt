// @vitest-environment jsdom
import { afterEach, expect, it, vi } from 'vitest'
import { click, render } from '../test/react'
import { PetPlay, type PetPlayProps } from './PetPlay'

const cleanups: (() => void)[] = []
afterEach(() => { cleanups.splice(0).forEach(fn => fn()); vi.unstubAllGlobals() })
function mount(patch: Partial<PetPlayProps> = {}) {
  const props: PetPlayProps = { petName: 'Bun', onReact: vi.fn(() => 'started' as const), ...patch }
  const ui = render(<PetPlay {...props} />); cleanups.push(ui.unmount)
  return { ...ui, props }
}
function button(container: Element, label: string) { return [...container.querySelectorAll('button')].find(node => node.textContent === label)! }

it.each(['away', 'sick'] as const)('explains unavailable viewing independently of %s', blocked => {
  const show = vi.fn(() => true)
  const ui = mount({ blocked, onShowRoom: show, showRoomUnavailable: "Choose Bun's room to see them." })
  click(button(ui.container, 'Spend time together'))
  const control = button(ui.container, 'Show room')
  expect(control.disabled).toBe(true)
  expect(document.getElementById(control.getAttribute('aria-describedby')!)!.textContent).toBe("Choose Bun's room to see them.")
  click(control)
  expect(show).not.toHaveBeenCalled()
  expect(ui.container.textContent).toContain('resting')
})

it.each(['away', 'sick'] as const)('announces a failed viewing callback even while %s', blocked => {
  const ui = mount({ blocked, onShowRoom: () => false })
  click(button(ui.container, 'Spend time together'))
  click(button(ui.container, 'Show room'))
  expect(ui.container.querySelector('[role="status"]')!.textContent).toContain('arrives')
})

it('offers Show room only while open and delegates without choosing a reaction', () => {
  const show = vi.fn(() => true)
  const ui = mount({ onShowRoom: show })
  expect(button(ui.container, 'Show room')).toBeUndefined()
  click(button(ui.container, 'Spend time together'))
  const control = button(ui.container, 'Show room')
  expect(control).toBeDefined()
  expect(control.getAttribute('aria-label')).toBe('Show room with Bun')
  click(control)
  expect(show).toHaveBeenCalledOnce()
  expect(ui.props.onReact).not.toHaveBeenCalled()
  expect(ui.container.querySelector('[role="status"]')!.textContent).toBe('')
  click(button(ui.container, 'Spend time together'))
  expect(button(ui.container, 'Show room')).toBeUndefined()
})

it('explains an unavailable room and clears the feedback after successful viewing', () => {
  const show = vi.fn().mockReturnValueOnce(false).mockReturnValue(true)
  const ui = mount({ onShowRoom: show })
  click(button(ui.container, 'Spend time together'))
  const control = button(ui.container, 'Show room')
  expect(control).toBeDefined()
  click(control)
  expect(ui.container.querySelector('[role="status"]')!.textContent).toContain('arrives')
  click(control)
  expect(ui.container.querySelector('[role="status"]')!.textContent).toBe('')
  expect(ui.props.onReact).not.toHaveBeenCalled()
})

it.each(['travel', 'away', 'sick'] as const)('allows room viewing while resting, but waits during %s', blocked => {
  const show = vi.fn(() => true)
  const ui = mount({ blocked, onShowRoom: show })
  click(button(ui.container, 'Spend time together'))
  const control = button(ui.container, 'Show room') as HTMLButtonElement
  expect(control).toBeDefined()
  expect(control.disabled).toBe(blocked === 'travel')
  click(control)
  expect(show).toHaveBeenCalledTimes(blocked === 'travel' ? 0 : 1)
  expect(ui.props.onReact).not.toHaveBeenCalled()
})

it('offers an explicit stretch without scheduling or duplicating a successful announcement', () => {
  vi.stubGlobal('requestAnimationFrame', vi.fn())
  const ui = mount()
  click(button(ui.container, 'Spend time together'))
  const stretch = button(ui.container, 'Stretch together')
  expect(stretch).toBeDefined()
  click(stretch)
  expect(ui.props.onReact).toHaveBeenCalledExactlyOnceWith('stretch')
  expect(requestAnimationFrame).not.toHaveBeenCalled()
  expect(ui.container.querySelector('[role="status"]')!.textContent).toBe('')
})

it('starts collapsed and sends only explicit choices without creating scheduling work', () => {
  vi.stubGlobal('requestAnimationFrame', vi.fn())
  const ui = mount()
  const toggle = button(ui.container, 'Spend time together')
  expect(toggle.getAttribute('aria-expanded')).toBe('false')
  expect(ui.container.querySelector('[hidden]')).not.toBeNull()
  click(toggle)
  click(button(ui.container, 'Wave'))
  expect(ui.props.onReact).toHaveBeenLastCalledWith('greet')
  click(button(ui.container, 'Gentle pats'))
  expect(ui.props.onReact).toHaveBeenLastCalledWith('affection')
  expect(button(ui.container, 'Happy hop')).toBeDefined()
  click(button(ui.container, 'Happy hop'))
  expect(ui.props.onReact).toHaveBeenLastCalledWith('hop')
  expect(ui.container.querySelector('[role="status"]')!.textContent).toBe('')
  expect(requestAnimationFrame).not.toHaveBeenCalled()
  click(toggle)
  expect(toggle.getAttribute('aria-expanded')).toBe('false')
})

it.each(['travel', 'away', 'sick'] as const)('explains and blocks play while %s', blocked => {
  const ui = mount({ blocked })
  click(button(ui.container, 'Spend time together'))
  for (const label of ['Wave', 'Gentle pats', 'Happy hop', 'Stretch together', 'Rest together']) {
    expect((button(ui.container, label) as HTMLButtonElement).disabled).toBe(true)
    click(button(ui.container, label))
  }
  expect(ui.props.onReact).not.toHaveBeenCalled()
  expect(ui.container.querySelector('.pet-play-note')!.textContent).toMatch(blocked === 'travel' ? /on the way/ : /resting/)
})

it('announces busy/unavailable feedback once and clears it after an accepted choice', () => {
  const results = ['busy', 'unavailable', 'started'] as const
  let count = 0
  const ui = mount({ onReact: () => results[count++] })
  click(button(ui.container, 'Spend time together'))
  const wave = button(ui.container, 'Wave')
  click(wave)
  expect(ui.container.querySelector('[role="status"]')!.textContent).toContain('moment')
  click(wave)
  expect(ui.container.querySelector('[role="status"]')!.textContent).toContain('arrive')
  click(wave)
  expect(ui.container.querySelector('[role="status"]')!.textContent).toBe('')
})


it('offers a quiet rest without starting a panel clock or duplicating successful speech', () => {
  vi.stubGlobal('requestAnimationFrame', vi.fn())
  const respond = vi.fn().mockReturnValueOnce('busy').mockReturnValue('started')
  const ui = mount({ onReact: respond })
  click(button(ui.container, 'Spend time together'))
  const rest = button(ui.container, 'Rest together')
  expect(rest).toBeDefined()
  click(rest)
  expect(respond).toHaveBeenLastCalledWith('rest')
  expect(ui.container.querySelector('[role="status"]')!.textContent).toContain('moment')
  click(rest)
  expect(ui.container.querySelector('[role="status"]')!.textContent).toBe('')
  expect(requestAnimationFrame).not.toHaveBeenCalled()
  expect([...ui.container.querySelectorAll('.pet-play-actions button')].map(node => node.textContent)).toEqual(['Wave', 'Gentle pats', 'Happy hop', 'Stretch together', 'Rest together'])
})

it('delegates Dance together without creating a second clock, write or success announcement', () => {
  vi.stubGlobal('requestAnimationFrame', vi.fn())
  const write = vi.spyOn(Storage.prototype, 'setItem')
  const ui = mount()
  click(button(ui.container, 'Spend time together'))
  const control = button(ui.container, 'Dance together')
  expect(control).toBeDefined()
  click(control)
  expect(ui.props.onReact).toHaveBeenCalledExactlyOnceWith('dance')
  expect(requestAnimationFrame).not.toHaveBeenCalled()
  expect(write).not.toHaveBeenCalled()
  expect(ui.container.querySelector('[role="status"]')!.textContent).toBe('')
  write.mockRestore()
})

it.each(['travel', 'away', 'sick'] as const)('declines a dance while %s without calling the resident', blocked => {

})
