import argparse
import datetime
import json
import os
from pathlib import Path
import shutil
import signal
import subprocess
import time

parser = argparse.ArgumentParser()
parser.add_argument('label')
parser.add_argument('--launch-mib', type=int, default=3200)
parser.add_argument('command', nargs=argparse.REMAINDER)
args = parser.parse_args()
command = args.command[1:] if args.command[:1] == ['--'] else args.command
if not command:
    parser.error('command required')
out = Path('docs/audits/2026-10-12-character-polish')
samples = []


def sample():
    available = int(next(line.split()[1] for line in Path('/proc/meminfo').read_text().splitlines() if line.startswith('MemAvailable:'))) * 1024
    row = {'elapsedSeconds': round(time.monotonic() - start, 3),
           'memAvailableBytes': available, 'tmpFreeBytes': shutil.disk_usage('/tmp').free}
    samples.append(row)
    return row


def live_group_members(pgid):
    members = []
    for path in Path('/proc').iterdir():
        if not path.name.isdigit():
            continue
        try:
            fields = (path / 'stat').read_text().rsplit(')', 1)[1].split()
            if int(fields[2]) == pgid and int(fields[3]) == pgid and fields[0] not in ('Z', 'X'):
                members.append(int(path.name))
        except (FileNotFoundError, ProcessLookupError):
            continue
    return sorted(members)


def signal_group(pgid, number):
    try:
        os.killpg(pgid, number)
    except ProcessLookupError:
        pass


class SupervisorSignal(Exception):
    def __init__(self, number):
        self.number = number


def interrupted(number, _frame):
    raise SupervisorSignal(number)


signal.signal(signal.SIGTERM, interrupted)
signal.signal(signal.SIGINT, interrupted)
start = time.monotonic()
first = sample()
reason = None
code = 125
child = None
with (out / f'{args.label}.log').open('w') as log:
    try:
        if first['memAvailableBytes'] < args.launch_mib * 1024**2 or first['tmpFreeBytes'] < 2 * 1024**3:
            reason = 'Insufficient launch headroom; no child started.'
            log.write(reason + '\n')
        else:
            child = subprocess.Popen(command, stdout=log, stderr=subprocess.STDOUT, start_new_session=True)
            while child.poll() is None:
                current = sample()
                if current['memAvailableBytes'] < 2 * 1024**3 or current['tmpFreeBytes'] < 2 * 1024**3:
                    reason = 'Resource reserve crossed; stopped only this job process group.'
                    code = 124
                    break
                time.sleep(1)
            if reason is None:
                code = child.returncode
    except SupervisorSignal as stopped:
        code = 128 + stopped.number
        reason = f'Supervisor received signal {stopped.number}; stopped only this job process group.'
    finally:
        signal.signal(signal.SIGTERM, signal.SIG_IGN)
        signal.signal(signal.SIGINT, signal.SIG_IGN)
        cleanup = {'processGroup': child.pid if child else None, 'termMembers': [], 'killMembers': [], 'remainingLiveMembers': []}
        if child is not None:
            cleanup['termMembers'] = live_group_members(child.pid)
            if cleanup['termMembers']:
                if reason is None:
                    reason = 'Direct child exited while owned descendants remained; stopped only this job process group.'
                    code = 126
                signal_group(child.pid, signal.SIGTERM)
                deadline = time.monotonic() + 3
                while live_group_members(child.pid) and time.monotonic() < deadline:
                    time.sleep(.05)
                cleanup['killMembers'] = live_group_members(child.pid)
                if cleanup['killMembers']:
                    signal_group(child.pid, signal.SIGKILL)
                    deadline = time.monotonic() + 2
                    while live_group_members(child.pid) and time.monotonic() < deadline:
                        time.sleep(.05)
                cleanup['remainingLiveMembers'] = live_group_members(child.pid)
                if cleanup['remainingLiveMembers']:
                    code = 127
                    reason += ' Live owned members remained after bounded escalation.'
            child.wait(timeout=2)
            cleanup['leaderReturnCode'] = child.returncode
        if reason:
            log.write('\n' + reason + '\n')
            log.flush()
        sample()
(out / f'{args.label}.exit').write_text(str(code) + '\n')
(out / f'{args.label}-resources.json').write_text(json.dumps({
    'utc': datetime.datetime.now(datetime.timezone.utc).isoformat(),
    'command': command, 'exitCode': code, 'resourceStopReason': reason,
    'launchMinimumMiB': args.launch_mib, 'runReserveMiB': 2048,
    'nodeOptions': os.environ.get('NODE_OPTIONS'),
    'cleanup': cleanup,
    'minimumObservedAvailableBytes': min(row['memAvailableBytes'] for row in samples),
    'method': 'Host available RAM and /tmp sampled about once per second; not a peak CPU/RSS or physical-device certification. Only this job process group may be stopped. SIGTERM/SIGINT or leader exit clean live owned session/group members independently of the leader, escalate after3s and preserve final sidecars. Zombies are excluded from live work.',
    'samples': samples,
}, indent=2) + '\n')
print(args.label, 'exit', code, 'minimum observed available MiB', round(min(row['memAvailableBytes'] for row in samples) / 1024**2, 1), reason or '')
raise SystemExit(code)
